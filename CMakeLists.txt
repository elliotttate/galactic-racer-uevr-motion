cmake_minimum_required(VERSION 3.20)
project(GalacticRacerMotion VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(motion_core STATIC src/Telemetry.cpp)
target_include_directories(motion_core PUBLIC src)
add_library(GalacticRacerMotion SHARED src/Plugin.cpp)
target_include_directories(GalacticRacerMotion PRIVATE vendor)
target_link_libraries(GalacticRacerMotion PRIVATE motion_core ws2_32)
target_compile_definitions(GalacticRacerMotion PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(motion_core PRIVATE /W4 /WX /utf-8)
  target_compile_options(GalacticRacerMotion PRIVATE /W4 /utf-8 /EHsc)
  set_property(TARGET GalacticRacerMotion motion_core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
enable_testing()
add_executable(motion_tests tests/MotionTests.cpp)
target_link_libraries(motion_tests PRIVATE motion_core)
if(MSVC)
  set_property(TARGET motion_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_test(NAME motion_tests COMMAND motion_tests)
